let contacts = JSON.parse(localStorage.getItem('group16_contacts') || '[]');
let editIndex = -1;

function saveContact(){
  let name = document.getElementById('name').value.trim();
  let phone = document.getElementById('phone').value.trim();
  let email = document.getElementById('email').value.trim();
  let address = document.getElementById('address').value.trim();

  if(!name || !phone || !email){ alert('Fill Name, Phone, Email'); return; }
  if(phone.length < 11){ alert('Phone must be 11 digits'); return; }
  if(!email.includes('@')){ alert('Invalid email'); return; }

  let obj = {name, phone, email, address};

  if(editIndex === -1){
    contacts.push(obj);
  } else {
    contacts[editIndex] = obj;
    editIndex = -1;
    document.getElementById('form-title').innerText='Add New Contact';
    document.getElementById('saveBtn').innerText='Save Contact';
    document.getElementById('cancelBtn').style.display='none';
  }

  localStorage.setItem('group16_contacts', JSON.stringify(contacts));
  clearForm();
  renderContacts();
}

function renderContacts(){
  let search = document.getElementById('search').value.toLowerCase();
  let list = document.getElementById('contactList');
  let filtered = contacts.filter(c => c.name.toLowerCase().includes(search) || c.phone.includes(search));
  
  document.getElementById('count').innerText = filtered.length + ' contacts';
  
  list.innerHTML = filtered.map((c, i) => `
    <div class="card">
      <div class="card-info">
        <b>${c.name}</b><br>
        📞 ${c.phone}<br>
        ✉️ ${c.email}<br>
        ${c.address ? '🏠 '+c.address : ''}
      </div>
      <div class="card-actions">
        <button class="edit" onclick="editContact(${contacts.indexOf(c)})">Edit</button>
        <button class="delete" onclick="deleteContact(${contacts.indexOf(c)})">Del</button>
      </div>
    </div>
  `).join('');
}

function editContact(i){
  let c = contacts[i];
  document.getElementById('name').value = c.name;
  document.getElementById('phone').value = c.phone;
  document.getElementById('email').value = c.email;
  document.getElementById('address').value = c.address;
  editIndex = i;
  document.getElementById('form-title').innerText='Edit Contact';
  document.getElementById('saveBtn').innerText='Update Contact';
  document.getElementById('cancelBtn').style.display='block';
  window.scrollTo(0,0);
}

function cancelEdit(){
  editIndex = -1;
  clearForm();
  document.getElementById('form-title').innerText='Add New Contact';
  document.getElementById('saveBtn').innerText='Save Contact';
  document.getElementById('cancelBtn').style.display='none';
}

function deleteContact(i){
  if(confirm('Delete this contact?')){
    contacts.splice(i,1);
    localStorage.setItem('group16_contacts', JSON.stringify(contacts));
    renderContacts();
  }
}

function clearForm(){
  document.getElementById('name').value='';
  document.getElementById('phone').value='';
  document.getElementById('email').value='';
  document.getElementById('address').value='';
}

renderContacts();
